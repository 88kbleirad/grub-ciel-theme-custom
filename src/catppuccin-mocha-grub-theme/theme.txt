# Boot menu — inside the HUD frame baked into the background.
# Canvas wrapper: GRUB 2.16 clamps a top-level boot_menu's height to its
# hardcoded 3-item minimum, so the visible-row count cannot be reduced by
# `height` alone. Nesting boot_menu in a canvas of the desired height works:
# visible rows = floor((H − item_padding) / (item_height + item_spacing)).
# 232 → exactly 2 rows (2×80 + 24 spacing + 24 padding); row 3 starts right
# at the canvas bottom edge, so no sliver. Scrolling on arrows.
+ canvas {
  left = 50%
  top = 29%
  width = 40%
  height = 232
  + boot_menu {
    left = 0
    top = 0
    width = 100%
    height = 232
    item_font = "BigBlueTerm437 Nerd Font Regular 28"
    selected_item_font = "BigBlueTerm437 Nerd Font Regular 28"
    item_color = "#cdd6f4"
    selected_item_color = "#cdd6f4"
    item_pixmap_style = "item_*.png"
    selected_item_pixmap_style = "select_*.png"
    item_height = 80
    item_spacing = 24
    item_padding = 24
    scrollbar = false
  }
}
